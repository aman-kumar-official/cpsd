Today we are going to see MQTT using HiveMQ broker with NODE JS

Go to https://www.hivemq.com/ and Click Menu  MQTT -> Free MQTT Broker -> Join with Google or any other means.

Once you login, click "Create a new cluster" -> Create Serverless Cluster (which is free upto 10GB) 

A Cluster will be created and click Manage Cluster, Note down the URL and the port number, something looks like this:
var options = {
    host: '7388baaacac5481a83afdbf63ca837e6.s1.eu.hivemq.cloud',
    port: 8883,
    protocol: 'mqtts',
    username: 'hivemq.webclient.1784261686844',
    password: 'h@U*0pXbIf3:x?n6BW9C'
}

Click Web client and there will be username/password section, create your own username and password or connect with autogenerated credentials which you can see as given above. This is my login credentials and url mentioned in the above code. 

Once this procedure is over, modify your details in the following code: Publisher.js and subscriber.js 

Pub.js
const mqtt = require("mqtt");
var options = {
    host: '7388baaacac5481a83afdbf63ca837e6.s1.eu.hivemq.cloud',
    port: 8883,
    protocol: 'mqtts',
    username: 'hivemq.webclient.1784261686844',
    password: 'h@U*0pXbIf3:x?n6BW9C'
}

var client = mqtt.connect(options);

client.on('connect', function () {
    console.log('Connected');
setInterval(function () {
var random1 = Math.random()*50;
console.log(random1);
if (random1 < 30)
client.publish('my/test/topic', 'Simple MQTT using HiveMQ: ' + random1.toString());
}),30000;
});

client.on('error', function (error) {
    console.log(error);
});

var mqtt = require('mqtt')

}

var client = mqtt.connect(options);
client.on('connect', function () {
